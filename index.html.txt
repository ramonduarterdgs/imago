    <script>
        const fileInput = document.getElementById('file-input');
        const photoPreview = document.getElementById('photo-preview');
        const editorSection = document.getElementById('editor-section');
        const uploadSection = document.getElementById('upload-section');
        const zoomSlider = document.getElementById('zoom-slider');
        const downloadBtn = document.getElementById('download-btn');
        const successArea = document.getElementById('success-area');
        const canvas = document.getElementById('export-canvas');
        const container = document.getElementById('container');

        let scale = 1;
        let baseScale = 1;
        let posX = 0, posY = 0;
        let isDragging = false;
        let startX, startY;
        let currentSizeMm = 32;

        window.setSize = (size, element) => {
            currentSizeMm = size;
            document.querySelectorAll('.size-card').forEach(c => c.classList.remove('active'));
            element.classList.add('active');
            successArea.classList.add('hidden');
        };

        window.resetAll = () => {
            fileInput.value = "";
            uploadSection.classList.remove('hidden');
            editorSection.classList.add('hidden');
            successArea.classList.add('hidden');
        };

        fileInput.addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(event) {
                    photoPreview.src = event.target.result;
                    photoPreview.onload = function() {
                        uploadSection.classList.add('hidden');
                        editorSection.classList.remove('hidden');
                        
                        const scaleW = 300 / photoPreview.naturalWidth;
                        const scaleH = 300 / photoPreview.naturalHeight;
                        baseScale = Math.max(scaleW, scaleH);
                        
                        scale = 1; posX = 0; posY = 0;
                        zoomSlider.value = 1;
                        updateTransform();
                    }
                };
                reader.readAsDataURL(file);
            }
        });

        // Mouse/Touch Events
        const startDrag = (e) => {
            isDragging = true;
            const clientX = e.touches ? e.touches[0].clientX : e.clientX;
            const clientY = e.touches ? e.touches[0].clientY : e.clientY;
            startX = clientX - posX;
            startY = clientY - posY;
            successArea.classList.add('hidden');
        };

        const moveDrag = (e) => {
            if (!isDragging) return;
            e.preventDefault();
            const clientX = e.touches ? e.touches[0].clientX : e.clientX;
            const clientY = e.touches ? e.touches[0].clientY : e.clientY;
            posX = clientX - startX;
            posY = clientY - startY;
            updateTransform();
        };

        const stopDrag = () => isDragging = false;

        container.addEventListener('mousedown', startDrag);
        window.addEventListener('mousemove', moveDrag);
        window.addEventListener('mouseup', stopDrag);
        container.addEventListener('touchstart', startDrag);
        window.addEventListener('touchmove', moveDrag);
        window.addEventListener('touchend', stopDrag);
        zoomSlider.addEventListener('input', (e) => { scale = parseFloat(e.target.value); updateTransform(); });

        function updateTransform() {
            const finalScale = baseScale * scale;
            photoPreview.style.width = (photoPreview.naturalWidth * finalScale) + 'px';
            photoPreview.style.height = (photoPreview.naturalHeight * finalScale) + 'px';
            const left = (300 - (photoPreview.naturalWidth * finalScale)) / 2 + posX;
            const top = (300 - (photoPreview.naturalHeight * finalScale)) / 2 + posY;
            photoPreview.style.left = left + 'px';
            photoPreview.style.top = top + 'px';
        }

        // Função para pegar a cor predominante das bordas
        function getEdgeColor() {
            const tempCanvas = document.getElementById('color-thief-canvas');
            const tCtx = tempCanvas.getContext('2d');
            tempCanvas.width = 50;
            tempCanvas.height = 50;
            tCtx.drawImage(photoPreview, 0, 0, 50, 50);
            
            // Pega amostras dos cantos
            const p1 = tCtx.getImageData(0, 0, 1, 1).data;
            const p2 = tCtx.getImageData(49, 0, 1, 1).data;
            const p3 = tCtx.getImageData(0, 49, 1, 1).data;
            const p4 = tCtx.getImageData(49, 49, 1, 1).data;
            
            // Média simples r,g,b
            const r = Math.round((p1[0] + p2[0] + p3[0] + p4[0]) / 4);
            const g = Math.round((p1[1] + p2[1] + p3[1] + p4[1]) / 4);
            const b = Math.round((p1[2] + p2[2] + p3[2] + p4[2]) / 4);
            
            return `rgb(${r},${g},${b})`;
        }

        downloadBtn.addEventListener('click', () => {
            const bleedMm = 10; // 1cm de sangria (total no diâmetro)
            const totalSizeMm = currentSizeMm + bleedMm;
            const dpi = 300;
            
            const totalPx = Math.round((totalSizeMm / 25.4) * dpi);
            const innerPx = Math.round((currentSizeMm / 25.4) * dpi);
            const bleedPx = (totalPx - innerPx) / 2;

            canvas.width = totalPx;
            canvas.height = totalPx;
            const ctx = canvas.getContext('2d');

            // 1. Preencher fundo com a cor da borda da imagem
            const edgeColor = getEdgeColor();
            ctx.fillStyle = edgeColor;
            ctx.beginPath();
            ctx.arc(totalPx/2, totalPx/2, totalPx/2, 0, Math.PI * 2);
            ctx.fill();

            // 2. Criar máscara de corte para a imagem principal (incluindo a área de sangria suave)
            ctx.save();
            ctx.beginPath();
            ctx.arc(totalPx/2, totalPx/2, totalPx/2, 0, Math.PI * 2);
            ctx.clip();

            const visualToReal = innerPx / 300;
            const finalScale = baseScale * scale * visualToReal;

            const drawWidth = photoPreview.naturalWidth * finalScale;
            const drawHeight = photoPreview.naturalHeight * finalScale;

            // Offset para centralizar considerando que o canvas agora é maior (tem sangria)
            const dx = (totalPx - drawWidth) / 2 + (posX * visualToReal);
            const dy = (totalPx - drawHeight) / 2 + (posY * visualToReal);

            ctx.drawImage(photoPreview, dx, dy, drawWidth, drawHeight);
            ctx.restore();

            // 3. Opcional: Desenhar uma linha guia bem fininha indicando onde termina o botton original
            // Descomente se quiser a guia visual:
            /*
            ctx.strokeStyle = 'rgba(255,255,255,0.2)';
            ctx.lineWidth = 1;
            ctx.beginPath();
            ctx.arc(totalPx/2, totalPx/2, innerPx/2, 0, Math.PI * 2);
            ctx.stroke();
            */

            const link = document.createElement('a');
            link.download = `imago_${currentSizeMm}mm_com_sangria.png`;
            link.href = canvas.toDataURL('image/png');
            link.click();

            successArea.classList.remove('hidden');
        });
    </script>
